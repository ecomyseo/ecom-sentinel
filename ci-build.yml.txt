name: build

on:
  push:
    branches: [ main ]
    tags: [ "v*" ]
  pull_request:
    branches: [ main ]

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: "8.0.x"

      - name: Build
        run: dotnet build -c Release --nologo

      - name: Self-test (detection logic)
        run: dotnet run -c Release -- selftest

      - name: Publish (self-contained win-x64, single file)
        if: startsWith(github.ref, 'refs/tags/')
        run: dotnet publish -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -o publish

      - name: Zip release
        if: startsWith(github.ref, 'refs/tags/')
        run: Compress-Archive -Path publish/* -DestinationPath EcomSentinel-${{ github.ref_name }}-win-x64.zip

      - name: Create GitHub Release
        if: startsWith(github.ref, 'refs/tags/')
        uses: softprops/action-gh-release@v2
        with:
          files: EcomSentinel-${{ github.ref_name }}-win-x64.zip
